cmake_minimum_required(VERSION 3.24)
project(EpochOnline LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIB OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ENABLE_ICE ON CACHE BOOL "" FORCE)
set(USE_STEAMWEBRTC OFF CACHE BOOL "" FORCE)
set(Protobuf_USE_STATIC_LIBS ON CACHE BOOL "" FORCE)
find_package(nlohmann_json CONFIG REQUIRED)
add_subdirectory(../../Source/GameNetworkingSockets ${CMAKE_BINARY_DIR}/gns)
add_executable(EpochOnline bridge.cpp)
target_compile_definitions(EpochOnline PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(EpochOnline PRIVATE GameNetworkingSockets_s nlohmann_json::nlohmann_json winhttp crypt32 bcrypt ws2_32)
add_executable(EpochTransportFixture EXCLUDE_FROM_ALL transport_fixture.cpp)
target_link_libraries(EpochTransportFixture PRIVATE ws2_32)
